gcc -std=c11 -Wall -Wextra -pedantic main.c model.c view.c list.c -o phonebook // to run simple doubly linked list based version
gcc -std=c11 -Wall -Wextra -pedantic -c list.c -o list.o //to compile list.c library to an object file list.o
ar rcs liblist.a list.o // to create the static library
gcc -std=c11 -Wall -Wextra -pedantic main.c model.c view.c -L. -llist -o phonebook // Compile and link the application

rm liblist.a // before compiling dynamic library

gcc -Wall -Wextra -fPIC -c list.c -o list.o //compile as position independent compiled
gcc -dynamiclib -o liblist.dylib list.o     //create the dynamic library
gcc -Wall -Wextra main.c model.c view.c -L. -llist -Wl,-rpath,@executable_path/list -o phonebook // link the application with dynamic library
otool -L phonebook // Verify that the application uses the dynamic library
nm -gU liblist.dylib //inspect exported symbols
